cmake_minimum_required(VERSION 3.22)
project(VispeakCaretBridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Qml)
find_package(KF6Config REQUIRED)
find_path(KWIN_INCLUDE_DIR main.h PATH_SUFFIXES kwin REQUIRED)
find_library(KWIN_LIBRARY NAMES kwin REQUIRED)
add_library(VispeakCaretBridge MODULE bridge.cpp)
target_include_directories(VispeakCaretBridge PRIVATE "${KWIN_INCLUDE_DIR}/.." "${KWIN_INCLUDE_DIR}")
target_link_libraries(VispeakCaretBridge PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Qml KF6::ConfigCore "${KWIN_LIBRARY}")
set_target_properties(VispeakCaretBridge PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/Vispeak/CaretBridge")
configure_file(qmldir "${CMAKE_BINARY_DIR}/Vispeak/CaretBridge/qmldir" COPYONLY)
include(CTest)
if(BUILD_TESTING)
    add_executable(caret-bridge-smoke smoke.cpp)
    target_link_libraries(caret-bridge-smoke PRIVATE Qt6::Core Qt6::Qml)
    add_test(NAME caret-bridge-smoke COMMAND caret-bridge-smoke "${CMAKE_BINARY_DIR}")
endif()
